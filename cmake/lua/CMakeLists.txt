cmake_minimum_required(VERSION 3.21)
set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_LIST_DIR}/../toolchains/host.cmake" CACHE FILEPATH "")
project(pslog_local_lua C)
include(../pslog_dependencies.cmake)
include(../pslog_local_runtime.cmake)
pslog_acquire_verified_archive(COMPONENT lua
    URL https://www.lua.org/ftp/lua-5.5.1.tar.gz
    SHA256 1c4b4068d67061f2a2231ad2b5422e77acea1487ea9890f6320af614f4373dce
    OUTPUT archive)
set(source "${CMAKE_CURRENT_BINARY_DIR}/source")
file(ARCHIVE_EXTRACT INPUT "${archive}" DESTINATION "${source}")
set(source "${source}/lua-5.5.1/src")
file(GLOB sources "${source}/*.c")
list(REMOVE_ITEM sources "${source}/lua.c" "${source}/luac.c" "${source}/onelua.c")
add_library(lua STATIC ${sources})
set_target_properties(lua PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(APPLE)
    target_compile_definitions(lua PRIVATE LUA_USE_MACOSX)
else()
    target_compile_definitions(lua PRIVATE LUA_USE_LINUX)
endif()
target_link_libraries(lua PUBLIC ${CMAKE_DL_LIBS} m)
add_executable(pslog_lua "${source}/lua.c")
target_link_libraries(pslog_lua PRIVATE lua)
pslog_configure_nonshipped_executable(pslog_lua)
set_target_properties(pslog_lua PROPERTIES ENABLE_EXPORTS ON
    BUILD_RPATH "${CMAKE_CURRENT_LIST_DIR}/../../build/lua-sdk/lib")
install(TARGETS lua ARCHIVE DESTINATION lib)
install(FILES "${source}/lua.h" "${source}/luaconf.h" "${source}/lauxlib.h" "${source}/lualib.h" DESTINATION include)
